
## PADS
This manual is for PADS (0.1.4, 2026-06-26) Personal Assistant and Deployment System

### Introduction
 PADS is a shell-based automation framework designed to simplify system tasks
and project scaffolding through concise command-line flags. PADS serves as a
wrapper around common system and development tasks,
providing a more streamlined and user-friendly interface.
PADS is designed to support the VCS, Workflow, and CI/CD of
[Operation Mindmap].

 - Repo:  https://github.com/padsRepo/pads
 - Blog:  https://padsrepo.github.io/pads/
 - Docs:  https://github.com/padsRepo/pads/wiki
 
 #### Before you start

Before you begin, make sure you meet these prerequisites:  

* Bash >= 5.2

#### Get started

1) Download: `git clone git+ssh://pi@rpi400.local/srv/repo/git/pads.git`

2) Install: `makepkg -si`

3) Add to /etc/pacman.conf:
```bash
[datasphere]
SigLevel = Optional TrustAll
Server = http://rpi400.local/db/core
```

### Examples

: In this example we make and start a Flask project named `project`.

`pads -Su`
: In this example we update the system and enter the password.

`pads -Su neofetch ...`
: In this example we install as many packages as we want seperated by spaces.
  You can use `pacman` or `yay`.

### Source Code
~~~bash
title="PADS"
subtitle="Personal Assistant and Deployment System"
version="0.1.4"
OPTSTRING="B:,C:,M:,U:,T:,D,x,v,h"
synopsis="<cmd> -[<module>] -[<argument>] [name...]"
syntax="${0##*/} -[${OPTSTRING}] -[arg]"
baseDir="$(dirname "$(readlink -f "${0%/*}")")"
description(){
  cat << EOF
 * ${title}
 * ${subtitle}
EOF
}
usage(){
  cat << EOF
  Synopsis: $synopsis
  Syntax: $syntax
  Usage:
    -B [-h] Build lib into PKGBUILD
    -C [-h] Create new projects
    -T [-h] Generate templates
    -U [-h] Update the repository with pkgname
    -D [-h] Dryrun Module
    -x      Set -x
    -v      Show Version
    -h      Show Usage
  Ex: pads -Mp padsutils > PKGBUILD | pads -Bp padsutils | pads -Ur padsutils
  See also: man pads, info pads
EOF
}
manual(){
  description
  usage
}
while getopts ${OPTSTRING} mod; do
  mod=${mod}
  case ${mod} in
    B) 
      shift $((OPTIND - 1))
      . "${baseDir}/${0##*/}/build.sh"
      case $OPTARG in
        "p" | "-p" | "--pkgbuild") pkgbuild ${@};;
        "m" | "-m" | "--metapkg") metapkg ${@};;
        "l" | "-l" | "--pyproject") pyProject ${@};;
        "t" | "-t" | "--test") dryrun ${@};;
        "h" | "-h" | "--help") usage;;
        ?) printf "$0: illegal option -- $OPTARG\n * ${0##*/} -${mod} -h for help\n"
      esac
      exit
      ;;
    C) 
      shift $((OPTIND - 1)); 
      . "${baseDir}/${0##*/}/create.sh"
      case $OPTARG in
        "a" | "-a" | "--architecture") architecture ${@};;
        "d" | "-d" | "--directory") createDir ${@};;
        "f" | "-f" | "--file") createFile ${@};;
        "t" | "-t" | "--test") dryrun ${@};;
        "h" | "-h" | "--help") usage;;
        ?) printf "$0: illegal option -- $OPTARG\n * ${0##*/} -${mod} -h for help\n"
      esac
      exit;;
    T) 
      shift $((OPTIND - 1)); 
      . "${baseDir}/${0##*/}/templates.sh"
      case $OPTARG in
        "p" | "-p" | "--pkgbuild") pkgbuildFile ${@};;
        "i" | "-i" | "--install") installFile;;
        "c" | "-c" | "--command") cmdFile ${@};;
        "m" | "-m" | "--metapkg") metapkg ${@};;
        "l" | "-l" | "--pyproject") pyprojectFile ${@};;
        "t") extract $@;;
        "x" | "-x" | "--pycli") pycliFile ${@};;
        "h" | "-h" | "--help") usage;;
        ?) printf "$0: illegal option -- $OPTARG\n * ${0##*/} -${mod} -h for help\n"
      esac
      exit;;
    D)
      ${baseDir}/bin/padsutils;
      echo $author;
      INFO "This is a test...";
      ;;
    U) 
      shift $((OPTIND - 1))
      . "${baseDir}/${0##*/}/update.sh"
      case $OPTARG in
        "r" | "-r" | "--repository") updateRepo ${@};;
        "h" | "-h" | "--help") usage;;
        ?) printf "$0: illegal option -- $OPTARG\n * ${0##*/} -${mod} -h for help\n"
      esac
      exit;;
    x) set -$mod;;
    v) echo "${version}"; exit;;
    h) usage; exit;;
    ?) echo " * ${0##*/} -h for help"; exit 2;;
    *) status "K18 Error...$(CODE 2)";;
  esac
done
[[ $# -eq 0 ]] && manual && exit
~~~
